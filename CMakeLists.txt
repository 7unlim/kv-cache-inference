cmake_minimum_required(VERSION 3.18)
project(paged_allocator LANGUAGES CXX)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_executable(allocator_test tests/allocator_test.cpp)
target_include_directories(allocator_test PRIVATE cpp)
target_compile_features(allocator_test PRIVATE cxx_std_17)
target_compile_options(allocator_test PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
  target_compile_options(allocator_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(allocator_test PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME allocator COMMAND allocator_test)
set_tests_properties(allocator PROPERTIES TIMEOUT 60)
